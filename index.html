    var playlist = [];

    // Web Audio API (LP 치직거림 노이즈 - 반드시 재생 중일 때만 소리 출력)

    var audioCtx = null;

    var crackleNode = null;

    var crackleGain = null;


    function initCrackle() {

      if (audioCtx) return;

      var AudioContext = window.AudioContext || window.webkitAudioContext;

      if (!AudioContext) return;

      audioCtx = new AudioContext();


      var bufferSize = 2 * audioCtx.sampleRate;

      var noiseBuffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);

      var output = noiseBuffer.getChannelData(0);


      for (var i = 0; i < bufferSize; i++) {

        var r = Math.random();

        if (r > 0.9985) {

          output[i] = (Math.random() * 2 - 1) * 0.45;

        } else {

          output[i] = (Math.random() * 2 - 1) * 0.015;

        }

      }


      crackleNode = audioCtx.createBufferSource();

      crackleNode.buffer = noiseBuffer;

      crackleNode.loop = true;


      crackleGain = audioCtx.createGain();

      // 초기 볼륨은 무조건 0으로 시작 (재생 상태가 될 때만 볼륨 반영)

      crackleGain.gain.value = 0;


      crackleNode.connect(crackleGain);

      crackleGain.connect(audioCtx.destination);

      crackleNode.start(0);

    }


    function updateCrackleVolume(val) {

      // 현재 음원이 재생 중(isPlaying === true)일 때만 볼륨을 변경

      if (crackleGain) {

        if (isPlaying) {

          crackleGain.gain.value = (val / 100) * 0.5;

        } else {

          crackleGain.gain.value = 0;

        }

      }

    }


    function setCrackleActive(active) {

      if (active) {

        initCrackle();

      }

      if (!audioCtx || !crackleGain) return;


      if (active) {

        if (audioCtx.state === 'suspended') {

          audioCtx.resume();

        }

        var val = document.getElementById('crackleVol').value / 100;

        crackleGain.gain.value = val * 0.5;

      } else {

        crackleGain.gain.value = 0;

      }

    }

    }

    function updateTurntableUI(playing) {

      } else {

        led.classList.remove('on');

        setCrackleActive(false);

      }

    }

    // 유튜브 URL 정밀 파싱 (일반, 단축, 뮤직, 쇼츠, 재생목록 모두 지원)

    function parseYouTubeUrl(url) {

      var listMatch = url.match(/[?&]list=([^#\&\?]+)/);

      var vidMatch = url.match(/(?:youtu\.be\/|youtube\.com\/(?:embed\/|v\/|watch\?v=|watch\?.+&v=|shorts\/)|music\.youtube\.com\/watch\?v=)([^#\&\?]{11})/);

      var videoId = vidMatch ? vidMatch[1] : null;

      var listId = listMatch ? listMatch[1] : null;

      return { videoId: videoId, listId: listId };

    }


    // 유튜브 곡 또는 재생목록 추가

    function addYoutubeLink() {

      var input = document.getElementById('ytUrlInput');

      var url = input.value.trim();

      var parsed = parseYouTubeUrl(url);


      if (!parsed.videoId && !parsed.listId) {

        alert('올바른 유튜브 또는 유튜브 뮤직 링크를 입력해 주세요.');

        return;

      }


      if (parsed.listId && !parsed.videoId) {

        playlist.push({

          type: 'yt_playlist',

          title: 'YouTube 재생목록 (' + parsed.listId.substring(0, 8) + '...)',

          listId: parsed.listId,

          thumb: ''

        });

      } else if (parsed.listId && parsed.videoId) {

        playlist.push({

          type: 'yt_playlist',

          title: 'YouTube 재생목록 스트리밍',

          listId: parsed.listId,

          videoId: parsed.videoId,

          thumb: 'https://img.youtube.com/vi/' + parsed.videoId + '/hqdefault.jpg'

        });

      } else {

        playlist.push({

          type: 'youtube',

          title: 'YouTube 트랙 (' + parsed.videoId + ')',

          videoId: parsed.videoId,

          thumb: 'https://img.youtube.com/vi/' + parsed.videoId + '/hqdefault.jpg'

        });

      }


      input.value = '';

      renderPlaylist();

      if (currentIndex === -1) {

        playTrack(playlist.length - 1);

      }

    }


    // 플레이리스트 렌더링

    function renderPlaylist() {

      var ul = document.getElementById('playlistEl');

      ul.innerHTML = '';

      for (var i = 0; i < playlist.length; i++) {

        (function(idx) {

          var item = playlist[idx];

          var li = document.createElement('li');

          if (idx === currentIndex) li.className = 'active';

          var badgeText = '내 파일';

          if (item.type === 'youtube') badgeText = 'YT 곡';

          if (item.type === 'yt_playlist') badgeText = 'YT 재생목록';

          li.innerHTML = '<span style="overflow:hidden;text-overflow:ellipsis;white-space:nowrap;">' + (idx + 1) + '. ' + item.title + '</span><span class="badge">' + badgeText + '</span>';

          li.onclick = function() { playTrack(idx); };

          ul.appendChild(li);

        })(i);

      }

    }


    // 유튜브 트랙 로드 실행 함수

    function loadYouTubeTrack(track) {

      document.getElementById('yt-wrapper').style.display = 'block';

      document.getElementById('yt-guide').innerText = '▶ 아래 유튜브 화면의 재생 버튼을 터치하면 LP판이 돌아갑니다';


      if (!ytReady || !ytPlayer) {

        pendingTrack = track;

        return;

      }


      if (track.type === 'youtube') {

        // iOS 호환을 위해 cueVideoById로 안전하게 먼저 불러온 뒤 재생 시도

        ytPlayer.cueVideoById(track.videoId);

        setTimeout(function() {

          if (ytPlayer && ytPlayer.playVideo) ytPlayer.playVideo();

        }, 300);

      } else if (track.type === 'yt_playlist') {

        ytPlayer.cuePlaylist({

          list: track.listId,

          listType: 'playlist'

        });

        setTimeout(function() {

          if (ytPlayer && ytPlayer.playVideo) ytPlayer.playVideo();

        }, 500);

      }

    }


    // 트랙 재생 실행

    function playTrack(index) {

      if (index < 0 || index >= playlist.length) return;


      currentIndex = index;

      var track = playlist[index];

      renderPlaylist();


      document.getElementById('trackTitle').innerText = track.title;

      var labelEl = document.getElementById('vinylLabel');


      if (track.type === 'local') {

        document.getElementById('trackSource').innerText = '로컬 오디오 파일 재생 중';

        document.getElementById('yt-wrapper').style.display = 'none';

        if (ytPlayer && ytReady && ytPlayer.stopVideo) {

          ytPlayer.stopVideo();

        }

        labelEl.style.backgroundImage = 'none';


        audioPlayer.src = track.url;

        audioPlayer.playbackRate = (currentRPM === 33) ? 1.0 : 1.25;

        audioPlayer.play();

        updateTurntableUI(true);

      } else if (track.type === 'youtube' || track.type === 'yt_playlist') {

        document.getElementById('trackSource').innerText = '유튜브 / 유튜브 뮤직 스트리밍 연동';

        audioPlayer.pause();

        updateTurntableUI(false); // 실제 유튜브 영상 재생이 시작될 때 회전 시작

        labelEl.style.backgroundImage = track.thumb ? ('url(' + track.thumb + ')') : 'none';

        loadYouTubeTrack(track);

      }

    }


    function togglePlay() {

      if (currentIndex === -1) {

        if (playlist.length > 0) playTrack(0);

        return;

      }

      var track = playlist[currentIndex];


      if (track.type === 'local') {

        if (audioPlayer.paused) {

          audioPlayer.play();

          updateTurntableUI(true);

        } else {

          audioPlayer.pause();

          updateTurntableUI(false);

        }

      } else if ((track.type === 'youtube' || track.type === 'yt_playlist') && ytReady && ytPlayer) {

        var state = ytPlayer.getPlayerState();

        if (state === 1) {

          ytPlayer.pauseVideo();

          updateTurntableUI(false);

        } else {

          ytPlayer.playVideo();

        }

      }

    }


    function nextTrack() {

      if (playlist.length === 0) return;

      var track = playlist[currentIndex];

      if (track && track.type === 'yt_playlist' && ytReady && ytPlayer && ytPlayer.nextVideo) {

        ytPlayer.nextVideo();

        return;

      }

      var next = (currentIndex + 1) % playlist.length;

      playTrack(next);

    }


    function prevTrack() {

      if (playlist.length === 0) return;

      var track = playlist[currentIndex];

      if (track && track.type === 'yt_playlist' && ytReady && ytPlayer && ytPlayer.previousVideo) {

        ytPlayer.previousVideo();

        return;

      }

      var prev = (currentIndex - 1 + playlist.length) % playlist.length;

      playTrack(prev);

    }


    audioPlayer.onended = function() {

      nextTrack();

    };

    function onYouTubeIframeAPIReady() {

      ytPlayer = new YT.Player('yt-player', {

        height: '170',

        width: '300',

        playerVars: {

          'playsinline': 1,

          'rel': 0,

          'origin': window.location.origin

        },

        events: {

          'onReady': function() {

            ytReady = true;

            if (pendingTrack) {

              loadYouTubeTrack(pendingTrack);

              pendingTrack = null;

            }

          },

          'onStateChange': function(event) {

            if (event.data === 1) {

              // 실제 영상이 재생 중(PLAYING)일 때만 LP 회전 및 치직거림 시작

              updateTurntableUI(true);

              document.getElementById('yt-guide').innerText = '♪ 유튜브 스트리밍 재생 중';

              if (ytPlayer.getVideoData) {

                var data = ytPlayer.getVideoData();

                if (data && data.title && currentIndex !== -1) {

                  document.getElementById('trackTitle').innerText = data.title;

                  playlist[currentIndex].title = data.title;

                  renderPlaylist();

                }

                if (data && data.video_id) {

                  document.getElementById('vinylLabel').style.backgroundImage = 'url(https://img.youtube.com/vi/' + data.video_id + '/hqdefault.jpg)';

                }

              }

            } else if (event.data === 2) {

              // 일시정지

              updateTurntableUI(false);

              document.getElementById('yt-guide').innerText = '⏸ 일시정지됨';

            } else if (event.data === 0) {

              // 재생 종료

              updateTurntableUI(false);

              var track = playlist[currentIndex];

              if (track && track.type !== 'yt_playlist') {

                nextTrack();

              }

            }

          },

          'onError': function(event) {

            updateTurntableUI(false);

            document.getElementById('yt-guide').innerText = '⚠ 이 영상은 저작권자의 외부 재생 제한(오류 ' + event.data + ')으로 재생할 수 없습니다. 다른 링크를 시도해 주세요.';

          }

        }

      });

    }
